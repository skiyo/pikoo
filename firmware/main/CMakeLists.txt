option(PIKOO_WAKE_VETO "Experimental local phonetic veto; requires private pack and separate acceptance" OFF)
option(PIKOO_WAKE_BENCHMARK "Synthetic-only wake timing image; no microphone, speaker or network" OFF)
option(PIKOO_COMMAND_BENCHMARK "Offline synthetic command-recognition probe; no microphone or network" OFF)
if(PIKOO_WAKE_VETO AND (PIKOO_WAKE_BENCHMARK OR PIKOO_COMMAND_BENCHMARK))
  message(FATAL_ERROR "Do not combine a normal voice verifier and an offline benchmark")
endif()
if(PIKOO_WAKE_BENCHMARK AND PIKOO_COMMAND_BENCHMARK)
  message(FATAL_ERROR "Select one diagnostic image")
endif()
set(PIKOO_MAIN_SRCS "main.cpp" "board_audio.cpp" "board_display.cpp" "wake_model.cpp")
if(PIKOO_WAKE_BENCHMARK)
  set(PIKOO_MAIN_SRCS "wake_benchmark.cpp" "board_display.cpp" "wake_model.cpp")
endif()
if(PIKOO_COMMAND_BENCHMARK)
  set(PIKOO_MAIN_SRCS "command_benchmark.cpp" "board_display.cpp")
endif()
idf_component_register(SRCS ${PIKOO_MAIN_SRCS} INCLUDE_DIRS "." REQUIRES pikoo_core pikoo_features pikoo_eyes nvs_flash esp_wifi esp_netif esp_event esp_timer json mbedtls esp_driver_i2s esp_driver_i2c esp_driver_spi esp_driver_gpio esp_driver_usb_serial_jtag esp_lcd app_update tcp_transport)
target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_17)
if(PIKOO_WAKE_VETO)
  set(PIKOO_WAKE_VETO_MODELS "" CACHE FILEPATH "Authorized private MN7/FST pack; never published")
  if(NOT PIKOO_WAKE_VETO_MODELS OR NOT EXISTS "${PIKOO_WAKE_VETO_MODELS}")
    message(FATAL_ERROR "Experimental veto requires explicit PIKOO_WAKE_VETO_MODELS")
  endif()
  idf_build_get_property(PIKOO_VETO_PYTHON PYTHON)
  execute_process(COMMAND "${PIKOO_VETO_PYTHON}" "${CMAKE_CURRENT_LIST_DIR}/../../tools/check_command_models.py" "${PIKOO_WAKE_VETO_MODELS}"
    RESULT_VARIABLE PIKOO_VETO_RESULT ERROR_VARIABLE PIKOO_VETO_ERROR OUTPUT_QUIET)
  if(NOT PIKOO_VETO_RESULT EQUAL 0)
    message(FATAL_ERROR "Invalid veto model pack: ${PIKOO_VETO_ERROR}")
  endif()
  configure_file("${PIKOO_WAKE_VETO_MODELS}" "${CMAKE_CURRENT_BINARY_DIR}/wake_veto_models.bin" COPYONLY)
  target_add_binary_data(${COMPONENT_LIB} "${CMAKE_CURRENT_BINARY_DIR}/wake_veto_models.bin" BINARY)
  target_sources(${COMPONENT_LIB} PRIVATE "wake_verifier.cpp")
  set(PIKOO_WAKE_VETO_CUTOFF "0.5" CACHE STRING "Versioned experimental veto cutoff; 0.1 or 0.5")
  if(NOT PIKOO_WAKE_VETO_CUTOFF MATCHES "^0\\.[15]$")
    message(FATAL_ERROR "Select an explicitly evaluated veto cutoff 0.1 or 0.5")
  endif()
  target_compile_definitions(${COMPONENT_LIB} PRIVATE PIKOO_WAKE_VETO=1 PIKOO_WAKE_VETO_CUTOFF=${PIKOO_WAKE_VETO_CUTOFF}f)
endif()

if(PIKOO_COMMAND_BENCHMARK)
  set(PIKOO_COMMAND_MODELS "" CACHE FILEPATH "Private locally packed ESP-SR model")
  set(PIKOO_COMMAND_CASES "" CACHE FILEPATH "Private synthetic PCM probe cases")
  foreach(kind MODELS CASES)
    if(NOT PIKOO_COMMAND_${kind} OR NOT EXISTS "${PIKOO_COMMAND_${kind}}")
      message(FATAL_ERROR "Command probe requires explicit PIKOO_COMMAND_${kind}")
    endif()
  endforeach()
  idf_build_get_property(PIKOO_COMMAND_PYTHON PYTHON)
  execute_process(COMMAND "${PIKOO_COMMAND_PYTHON}" "${CMAKE_CURRENT_LIST_DIR}/../../tools/check_command_models.py" "${PIKOO_COMMAND_MODELS}"
    RESULT_VARIABLE PIKOO_COMMAND_PACK_RESULT OUTPUT_VARIABLE PIKOO_COMMAND_PACK_INFO ERROR_VARIABLE PIKOO_COMMAND_PACK_ERROR)
  if(NOT PIKOO_COMMAND_PACK_RESULT EQUAL 0)
    message(FATAL_ERROR "Invalid command model pack: ${PIKOO_COMMAND_PACK_ERROR}")
  endif()
  configure_file("${PIKOO_COMMAND_MODELS}" "${CMAKE_CURRENT_BINARY_DIR}/command_models.bin" COPYONLY)
  configure_file("${PIKOO_COMMAND_CASES}" "${CMAKE_CURRENT_BINARY_DIR}/command_cases.bin" COPYONLY)
  target_add_binary_data(${COMPONENT_LIB} "${CMAKE_CURRENT_BINARY_DIR}/command_models.bin" BINARY)
  target_add_binary_data(${COMPONENT_LIB} "${CMAKE_CURRENT_BINARY_DIR}/command_cases.bin" BINARY)
  return()
endif()
set(PIKOO_WAKE_MODEL "" CACHE FILEPATH "Local wake model; restricted models must not be published")
if(NOT PIKOO_WAKE_MODEL OR NOT EXISTS "${PIKOO_WAKE_MODEL}")
  message(FATAL_ERROR "No wake weights are distributed. Build with -DPIKOO_WAKE_MODEL=/absolute/path/to/authorized-model.tflite; see docs/getting-started.md")
endif()
file(SIZE "${PIKOO_WAKE_MODEL}" PIKOO_WAKE_MODEL_BYTES)
set(PIKOO_WAKE_MAX_BYTES "524288" CACHE STRING "DualEye experimental wake flash budget; arena and timing require hardware validation")
if(NOT PIKOO_WAKE_MAX_BYTES MATCHES "^[0-9]+$" OR PIKOO_WAKE_MAX_BYTES LESS 65536 OR PIKOO_WAKE_MAX_BYTES GREATER 524288)
  message(FATAL_ERROR "DualEye wake model budget must be 65536..524288 bytes")
endif()
if(PIKOO_WAKE_MODEL_BYTES GREATER PIKOO_WAKE_MAX_BYTES)
  message(FATAL_ERROR "Wake model exceeds the configured DualEye flash budget")
endif()
